## ⚠️ КРИТИЧЕСКАЯ НАХОДКА ПО ПУНКТУ A (clipboard)

Изучив исходник установленного `flet==0.81.0`
(`flet/controls/services/clipboard.py:36-51`):

```python
async def set_image(self, value: bytes) -> None:
    if not (self.page.web or self.page.platform.is_mobile()):
        raise FletUnsupportedPlatformException(
            "set_image is not supported on this platform"
        )
    await self._invoke_method("set_image", {"data": value})
```

DASMixer запускается как desktop-приложение (`ft.app(target=main)` в
`gui/app.py:27`), поэтому `page.web == False`, `page.platform` —
`WINDOWS`/`LINUX`/`MACOS`, `is_mobile() == False` ⇒ **вызов
`await ft.Clipboard().set_image(img_bytes)` всегда выбрасывает
`FletUnsupportedPlatformException`**. Документация Flet подтверждает:
"`set_image` is supported on Android, iOS, Web". На desktop в бинарнике
есть `pasteboard_plugin.dll`, но Python-сторона не пробрасывает вызов.

### Предлагаемое решение (требуется подтверждение разработчика)

Реализовать helper `gui/components/clipboard.py` со стратегией
**flet-first → native fallback**:

1. Сначала `await ft.Clipboard().set_image(png_bytes)` (будет работать
   на web/mobile и в будущих версиях flet, если поддержка desktop
   появится).
2. При `FletUnsupportedPlatformException` — платформенный fallback:
   - **Windows:** `pythonnet` (уже зависимость dasmixer-gui):
     `System.Windows.Forms.Clipboard.SetDataObject(DataObject)`,
     форматы `PNG` + `Bitmap`, в STA-потоке `.NET`.
     **Проверено экспериментально** в окружении проекта — работает.
   - **Linux:** subprocess `wl-copy --type image/png` (Wayland) или
     `xclip -selection clipboard -t image/png` (X11). Требуется наличие
     `wl-clipboard` или `xclip`; иначе понятная ошибка.
   - **macOS:** `osascript` + временный PNG-файл
     (`set the clipboard to (read (POSIX file ...) as «class PNGf»)`).
- Все блокирующие вызовы — через `loop.run_in_executor`, чтобы не
  замораживать UI.
- Любой Exception логируется через `logger.exception()` и
  пользователю показывается красный snack (по требованию).

**Альтернативы (не рекомендую):**
- Пакет `pyperclipimg` — добавляет Pillow + pywin32/pyobjc, что уже
  покрывается pythonnet на Windows и системными командами на Linux/macOS.
- `ft.Clipboard().set_files([temp_png_path])` — кладёт **файл** в
  clipboard, а не изображение; вставка в mspaint/GIMP не сработает.

Если разработчик предпочтёт без native fallback — пункт A не будет
работать на desktop. Нужен явный выбор (см. «Открытые вопросы»).

```python
async def copy_png_to_clipboard(page: ft.Page, png_bytes: bytes) -> None:
    """
    Скопировать PNG-байты в системный clipboard.

    Стратегия: flet.Clipboard.set_image -> native fallback.
    Raises RuntimeError при невозможности скопировать.
    """
```

**Структура (кросс-платформенный fallback):**

```python
import asyncio, sys, shutil, subprocess, tempfile
from pathlib import Path
import flet as ft
from flet.controls.exceptions import FletUnsupportedPlatformException
from dasmixer.utils import logger

async def copy_png_to_clipboard(page, png_bytes):
    # 1) Flet path (web/mobile/future desktop)
    try:
        await ft.Clipboard().set_image(png_bytes)
        return
    except FletUnsupportedPlatformException:
        pass  # desktop — идём в native fallback
    except Exception:
        logger.exception("ft.Clipboard().set_image() unexpected error")
        # не возвращаемся — пробуем fallback

    # 2) Native fallback
    loop = asyncio.get_event_loop()
    try:
        await loop.run_in_executor(None, _copy_native, png_bytes)
    except Exception as ex:
        logger.exception(ex)
        raise RuntimeError(str(ex)) from ex

def _copy_native(png_bytes: bytes) -> None:
    if sys.platform == "win32":
        _win_copy(png_bytes)
    elif sys.platform == "darwin":
        _mac_copy(png_bytes)
    else:  # linux и прочее
        _linux_copy(png_bytes)

def _win_copy(png_bytes: bytes) -> None:
    import clr
    clr.AddReference("System.Windows.Forms")
    clr.AddReference("System.Drawing")
    from System.Threading import (Thread as NetThread,
                                  ThreadStart, ApartmentState)
    from System.Windows.Forms import Clipboard, DataObject
    from System.Drawing import Image as DImage
    from System.IO import MemoryStream
    from System import Array, Byte

    err = {}
    def worker():
        try:
            do = DataObject()
            do.SetData("PNG", False, MemoryStream(Array[Byte](png_bytes)))
            img = DImage.FromStream(MemoryStream(Array[Byte](png_bytes)))
            do.SetImage(img)
            Clipboard.SetDataObject(do, True)
        except Exception as e:
            err["e"] = e
    t = NetThread(ThreadStart(worker))
    t.SetApartmentState(ApartmentState.STA)
    t.Start(); t.Join()
    if err:
        raise err["e"]

def _linux_copy(png_bytes: bytes) -> None:
    wl = shutil.which("wl-copy")
    if wl:
        subprocess.run([wl, "--type", "image/png"],
                       input=png_bytes, check=True)
        return
    xc = shutil.which("xclip")
    if xc:
        subprocess.run([xc, "-selection", "clipboard",
                        "-t", "image/png"],
                       input=png_bytes, check=True)
        return
    raise RuntimeError(
        "Neither 'wl-copy' nor 'xclip' is installed. "
        "Install 'wl-clipboard' (Wayland) or 'xclip' (X11)."
    )

def _mac_copy(png_bytes: bytes) -> None:
    import tempfile, os
    fd, p = tempfile.mkstemp(suffix=".png")
    try:
        with os.fdopen(fd, "wb") as f:
            f.write(png_bytes)
        # «class PNGf» — литералы «» набраны как \xab / \xbb
        script = (
            "set the clipboard to "
            f"(read (POSIX file \"{p}\") as \xabclass PNGf\xbb)"
        )
        subprocess.run(["osascript", "-e", script], check=True)
    finally:
        try: os.unlink(p)
        except OSError: pass
```

> Примечание: литералы `«class PNGf»` в AppleScript нужно набирать
> гуиллеметами; в Python-строке это `\xab`/`\xbb`. Реализация должна
> сохранять кодировку UTF-8 для файла.

## 2. Пункт B — Переиспользуемый компонент `ColorPickerField`

### 2.1. Новый модуль `gui/components/color_picker.py`

**Путь:** `dasmixer-gui/src/dasmixer/gui/components/color_picker.py`

**Концепция:**
- Внешний API работает с форматом **`#RRGGBB`** (с `#`, alpha всегда
  `FF`, в БД/конфиге уже хранится так — подтверждено разработчиком и
  кодом `constants.DEFAULT_COLORS`).
- Внутренне взаимодействие со `SlidePicker` — в формате flet
  **`#AARRGGBB`** (alpha всегда `FF`).
- Два режима: `compact` (для Settings, пикер открывается в диалоге по
  клику на preview) и `full` (для диалогов Group/Tool, пикер виден
  сразу). Решение (C), подтверждённое разработчиком.

**Константы/хелперы:**

```python
import re
import flet as ft
from flet_color_pickers import SlidePicker

HEX_COLOR_RE = re.compile(r"^#[0-9a-fA-F]{6}$")
_RGB_RE = re.compile(r"^[0-9a-fA-F]{6}$")

def _to_argb(rrggbb: str) -> str:
    """'#RRGGBB' -> '#FFRRGGBB' (flet-color-pickers format)."""
    return f"#ff{rrggbb[1:].lower()}"

def _from_argbb(s: str) -> str:
    """'#AARRGGBB' или '#RRGGBB' -> '#RRGGBB' uppercase."""
    h = s.lstrip("#")
    rgb = h[-6:] if len(h) >= 6 else h
    return f"#{rgb.upper()}"

def _is_valid_hex_rrggbb(s: str) -> bool:
    h = s.lstrip("#")
    return bool(_RGB_RE.match(h))
```

**Класс `ColorPickerField`:**

```python
class ColorPickerField(ft.Container):
    """
    Переиспользуемый выбор цвета на базе flet_color_pickers.SlidePicker.

    value всегда '#RRGGBB'. Режимы:
      compact=True  — preview(кликабельный) + TextField (+ delete btn).
                      Пикер открывается в AlertDialog по клику на preview.
      compact=False — preview + SlidePicker + TextField в одну строку.
    """

    def __init__(
        self,
        value: str = "#888888",
        *,
        label: str = "Color (hex)",
        compact: bool = False,
        show_delete: bool = False,
        on_delete: callable = None,
        on_change: callable = None,
        preview_size: int = 50,
        slider_width: int = 180,
    ):
        super().__init__()
        self._compact = compact
        self._show_delete = show_delete
        self._on_delete_cb = on_delete
        self._on_change_cb = on_change
        self._preview_size = preview_size
        self._slider_width = slider_width
        self._picker: SlidePicker | None = None
        self._syncing = False  # защита от feedback loop
        self._value = _normalize(value)

        self._preview = ft.Container(
            width=preview_size, height=preview_size,
            border_radius=5, bgcolor=self._value,
            border=ft.border.all(1, ft.Colors.GREY_400),
        )
        if not compact:
            # делаем preview кликабельным только в compact-режиме
            self._preview.on_click = None
        else:
            self._preview.on_click = self._on_preview_click

        self._field = ft.TextField(
            label=label,
            value=self._value[1:],          # показываем без '#'
            max_length=6,
            width=100,
            height=60,
            hint_text="e.g., FF0000",
            on_change=self._on_text_change,
            on_blur=self._on_text_blur,
        )

        content_controls = [self._preview]
        if not compact:
            self._picker = SlidePicker(
                color=_to_argb(self._value),
                show_indicator=False,
                enable_alpha=False,
                display_thumb_color=True,
                on_color_change=self._on_picker_change,
                slider_size=ft.Size(width=slider_width, height=20),
            )
            content_controls.append(self._picker)
        content_controls.append(self._field)

        if show_delete:
            content_controls.append(ft.IconButton(
                icon=ft.Icons.DELETE_OUTLINE,
                tooltip="Remove color",
                on_click=lambda e: (self._on_delete_cb() if self._on_delete_cb else None),
            ))

        self.content = ft.Row(
            content_controls,
            spacing=10,
            vertical_alignment=ft.CrossAxisAlignment.CENTER,
            height=80 if not compact else None,
        )

    # --- Внешний API ---

    @property
    def value(self) -> str:
        """Возвращает '#RRGGBB' (нормализованно, uppercase)."""
        return self._value

    @value.setter
    def value(self, v: str) -> None:
        """Программно выставить цвет (без вызова on_change)."""
        nv = _normalize(v)
        self._value = nv
        self._field.value = nv[1:]
        self._preview.bgcolor = nv
        if self._picker is not None:
            self._syncing = True
            self._picker.color = _to_argb(nv)
            self._syncing = False
        if self.page:
            self._update_all()

    @property
    def is_valid(self) -> bool:
        return _is_valid_hex_rrggbb(self._field.value or "")

    def set_error(self, msg: str | None = None) -> None:
        if msg:
            self._field.error_text = msg
            self._field.border_color = ft.Colors.RED
        else:
            self._field.error_text = None
            self._field.border_color = None
        if self.page:
            try: self._field.update()
            except Exception: pass

    # --- Внутренние обработчики ---

    def _on_picker_change(self, e: ft.ControlEvent) -> None:
        if self._syncing:
            return
        rgb = _from_argbb(e.data)              # '#RRGGBB'
        self._value = rgb
        self._field.value = rgb[1:]             # без '#'
        self._preview.bgcolor = rgb
        if self.page:
            self._update_all()
        if self._on_change_cb:
            self._on_change_cb(rgb)

    def _on_text_change(self, e: ft.ControlEvent) -> None:
        raw = (e.control.value or "").strip()
        if _is_valid_hex_rrggbb(raw):
            rgb = f"#{raw.upper()}"
            self._value = rgb
            self._preview.bgcolor = rgb
            if self._picker is not None:
                self._syncing = True
                self._picker.color = _to_argb(rgb)
                self._syncing = False
            e.control.border_color = None
        else:
            self._preview.bgcolor = "#888888"
        if self.page:
            try:
                self._preview.update()
                e.control.update()
                if self._picker is not None:
                    self._picker.update()
            except Exception:
                pass
        if self._on_change_cb:
            self._on_change_cb(self.value)

    def _on_text_blur(self, e: ft.ControlEvent) -> None:
        raw = (e.control.value or "").strip()
        if not _is_valid_hex_rrggbb(raw):
            e.control.border_color = ft.Colors.RED
            if self.page:
                try: e.control.update()
                except Exception: pass

    def _on_preview_click(self, e: ft.ControlEvent) -> None:
        """Открыть пикер в AlertDialog (только compact-режим)."""
        if not self.page:
            return
        picker = SlidePicker(
            color=_to_argb(self._value),
            show_indicator=False,
            enable_alpha=False,
            display_thumb_color=True,
            on_color_change=self._on_dialog_picker_change,
            slider_size=ft.Size(width=self._slider_width, height=20),
        )
        self._dialog_picker = picker  # ссылка для обновления
        dlg = ft.AlertDialog(
            title=ft.Text("Pick color"),
            content=ft.Container(
                content=ft.Column([picker], tight=True),
                padding=10,
            ),
            actions=[
                ft.TextButton("Done",
                    on_click=lambda ev: self._close_picker_dialog(dlg)),
            ],
        )
        self.page.overlay.append(dlg)
        dlg.open = True
        self.page.update()

    def _on_dialog_picker_change(self, e: ft.ControlEvent) -> None:
        rgb = _from_argbb(e.data)
        self._value = rgb
        self._field.value = rgb[1:]
        self._preview.bgcolor = rgb
        if self.page:
            self._update_all()
        if self._on_change_cb:
            self._on_change_cb(rgb)

    def _close_picker_dialog(self, dlg: ft.AlertDialog) -> None:
        dlg.open = False
        if self.page:
            self.page.update()

    def _update_all(self) -> None:
        for c in (self._preview, self._field, self._picker):
            if c is None:
                continue
            try:
                c.update()
            except Exception:
                pass


def _normalize(v: str) -> str:
    """Любой ввод -> '#RRGGBB' uppercase; невалидный -> '#888888'."""
    if not v:
        return "#888888"
    h = v.lstrip("#")
    if len(h) == 8:
        h = h[-6:]
    if len(h) == 6 and _RGB_RE.match(h):
        return f"#{h.upper()}"
    return "#888888"
```

   ```python
       self.page.update()
       return
   ```
